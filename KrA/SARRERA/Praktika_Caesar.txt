void print_string (unsigned char a[], int len, int despl)
{
        int ken;
        for (int i=0; i<len; i++)
        {
                if (a[i] == ',' || a[i] == '.' || a[i] == ' ') {
                        ken = a[i];
                }
                else {
                        ken = a[i] - despl;
                        if (a[i] < 91) {
                                if (ken < 65) {
                                        ken = ken + 26;
                                }
                        }
                        else {
                                if (ken < 97) {
                                        ken = ken + 26;
                                }
                        }
                }
                printf("%c", ken);
        }

        printf("\nErroketa: %d\n", despl);
